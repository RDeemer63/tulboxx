import { Request, Response, NextFunction } from 'express';
import { ZodError, ZodIssue } from 'zod';

/**
 * @interface ErrorDetails
 * @description Structure for providing more detailed error information, often used for validation errors.
 * Can be a string for a simple detail, or an object for structured details (e.g., field-specific validation issues).
 */
export type ErrorDetails = string | Record<string, any> | ZodIssue[] | undefined;

/**
 * @class AppError
 * @extends Error
 * @description Base class for custom application errors.
 *
 * @param {string} message - Error message.
 * @param {number} statusCode - HTTP status code.
 * @param {boolean} [isOperational=true] - Indicates if the error is operational (known, handled) vs. a programming error.
 * @param {ErrorDetails} [details] - Optional additional details about the error (e.g., validation specifics).
 */
export class AppError extends Error {
  public readonly statusCode: number;
  public readonly isOperational: boolean;
  public readonly details?: ErrorDetails;
  public readonly status: string; // 'fail' for 4xx, 'error' for 5xx

  constructor(message: string, statusCode: number, isOperational: boolean = true, details?: ErrorDetails) {
    super(message);
    this.statusCode = statusCode;
    this.isOperational = isOperational;
    this.details = details;
    this.status = `${statusCode}`.startsWith('4') ? 'fail' : 'error';

    // Ensure the prototype chain is correctly set for custom errors
    Object.setPrototypeOf(this, new.target.prototype);

    // Capture stack trace, excluding constructor call from it
    Error.captureStackTrace(this, this.constructor);
  }
}

/**
 * Formats Zod validation errors into a more structured details object.
 * @param zodError - The ZodError instance.
 * @returns A structured object of validation errors or undefined.
 */
const formatZodErrorDetails = (zodError: ZodError): Record<string, string[]> | undefined => {
  const fieldErrors = zodError.flatten().fieldErrors;
  if (Object.keys(fieldErrors).length > 0) {
    return fieldErrors as Record<string, string[]>; // Cast because fieldErrors can have undefined values
  }
  // Handle form errors (errors not specific to a field)
  const formErrors = zodError.flatten().formErrors;
  if (formErrors.length > 0) {
    return { _form: formErrors };
  }
  return undefined;
};

 * Helper function to create a standardized error response for use in places
 * where middleware might not be directly applicable (e.g., WebSocket error handling).
 *
 * @param error The error object.
 * @returns A standardized error response object.
 */
export const formatErrorResponse = (error: Error | AppError | ZodError): Record<string, any> => {
  const isDev = process.env.NODE_ENV === 'development';
  
  if (error instanceof ZodError) {
    return {
      status: 'fail',
      message: 'Input validation failed.',
      statusCode: 400,
      details: formatZodErrorDetails(error),
      ...(isDev && { stack: error.stack }),
    };
  }
  
  if (error instanceof AppError) {
    return {
      status: error.status,
      message: error.message,
      statusCode: error.statusCode,
      ...(error.details && { details: error.details }),
      ...(isDev && error.isOperational && { stack: error.stack }), // Stack for operational errors in dev
      ...(isDev && !error.isOperational && { stack: error.stack }), // Stack for non-op errors in dev
    };
  }

  // Generic unhandled error
  return {
    status: 'error',
    message: isDev ? error.message : 'An unexpected internal server error occurred.',
    statusCode: 500,
    ...(isDev && { stack: error.stack }),
  };
};
