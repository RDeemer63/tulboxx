async function throwIfResNotOk(res: Response) {
  if (!res.ok) {
    if (res.status === 401) {
      logout();
      return;
    }
    const text = (await res.text()) || res.statusText;
    throw new Error(`${res.status}: ${text}`);
  }
}

export async function apiRequest(
  method: string,
  url: string,
  data?: unknown | undefined,
): Promise<Response> {
  const headers: Record<string, string> = {
    "Content-Type": "application/json",
    ...getAuthHeaders()

  const res = await fetch(url, {

    headers,
    body: data ? JSON.stringify(data) : undefined,
    credentials: "include",

app.get("/api/customers", authenticateToken, async (req, res) => {
  try {
    const { page, limit, search } = paginationSchema.parse(req.query);
    const offset = (page - 1) * limit;

    let query = storage.db.select().from(storage.customers);
    
    // Add search if provided
    if (search) {
      query = query.where(
        or(
          ilike(storage.customers.firstName, `%${search}%`),
          ilike(storage.customers.lastName, `%${search}%`),
          ilike(storage.customers.email, `%${search}%`)
        )
      );
    }

    // Get paginated results
    const customers = await query
      .orderBy(desc(storage.customers.createdAt))
      .limit(limit)
      .offset(offset);

    // Get total count for pagination
    const totalQuery = storage.db.select({ count: count() }).from(storage.customers);
    const [{ count: total }] = search 
      ? await totalQuery.where(
          or(
            ilike(storage.customers.firstName, `%${search}%`),
            ilike(storage.customers.lastName, `%${search}%`),
            ilike(storage.customers.email, `%${search}%`)
          )
        )
      : await totalQuery;

    res.json({
      data: customers,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
        hasNext: page * limit < total,
        hasPrev: page > 1,
      },
    });
  } catch (error) {
    console.error("Error fetching customers:", error);
    res.status(500).json({ message: "Failed to fetch customers" });
  }
});
